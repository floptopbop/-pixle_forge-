min : (val > max ? max : val); /** Convert RGBA Uint8ClampedArray to grayscale Float32Array [0–255] */ Core.toGray = function(data, w, h) { const out = new Float32Array(w * h); for (let i = 0, p = 0; i < data.length; i += 4, p++) { // Human eye weighting (BT.601 standard) out[p] = data[i] * 0.299 + data[i + 1] * 0.587 + data[i + 2] * 0.114; } return out; }; /** 1D sliding-window box blur (In-place mutation to target buffer) */ Core.boxBlur1D = function(src, out, w, h, radius, horizontal) { const size = radius * 2 + 1; if (horizontal) { for (let y = 0; y < h; y++) { let acc = 0; const row = y * w; for (let x = -radius; x <= radius; x++) { acc += src[row + FastClamp(x, 0, w - 1)]; } for (let x = 0; x < w; x++) { out[row + x] = acc / size; const addX = FastClamp(x + radius + 1, 0, w - 1); const subX = FastClamp(x - radius, 0, w - 1); acc += src[row + addX] - src[row + subX]; } } } else { for (let x = 0; x < w; x++) { let acc = 0; for (let y = -radius; y <= radius; y++) { acc += src[FastClamp(y, 0, h - 1) * w + x]; } for (let y = 0; y < h; y++) { out[y * w + x] = acc / size; const addY = FastClamp(y + radius + 1, 0, h - 1); const subY = FastClamp(y - radius, 0, h - 1); acc += src[addY * w + x] - src[subY * w + x]; } } } }; /** Optimized Gaussian approximation using double-buffer ping-ponging */ Core.gaussianApprox = function(src, w, h, radius) { if (radius <= 0) return src.slice(); const r = Math.max(1, Math.round(radius)); // Allocate only once! Returns {gx, gy, mag, angle} */ Core.sobel = function(gray, w, h) { const gx = new Float32Array(w * h); const gy = new Float32Array(w * h); const mag = new Float32Array(w * h); const angle = new Float32Array(w * h);
